(() => {
  // ../coding-agent/src/tools/browser/relay/protocol.ts
  var OVERLAY_HOST_TAG = "omp-overlay";
  // extension/teach/element.ts
  var SELECTOR_ATTRIBUTES = ["data-testid", "data-test", "data-qa", "name", "aria-label"];
  var NATIVE_ROLES = {
    article: "article",
    button: "button",
    details: "group",
    dialog: "dialog",
    fieldset: "group",
    form: "form",
    img: "img",
    li: "listitem",
    main: "main",
    meter: "meter",
    nav: "navigation",
    ol: "list",
    option: "option",
    output: "status",
    progress: "progressbar",
    summary: "button",
    table: "table",
    textarea: "textbox",
    ul: "list"
  };
  var CONTENT_ROLES = {
    button: true,
    link: true,
    tab: true,
    option: true,
    menuitem: true,
    menuitemcheckbox: true,
    menuitemradio: true,
    radio: true,
    checkbox: true,
    switch: true,
    heading: true,
    treeitem: true
  };
  function normalize(text) {
    return text?.replace(/\s+/g, " ").trim() ?? "";
  }
  function roleFor(element) {
    const explicit = normalize(element.getAttribute("role")).split(" ", 1)[0];
    if (explicit)
      return explicit;
    const tag = element.localName;
    switch (tag) {
      case "input": {
        const type = element.getAttribute("type")?.toLowerCase();
        switch (type) {
          case "button":
          case "image":
          case "reset":
          case "submit":
            return "button";
          case "checkbox":
            return "checkbox";
          case "radio":
            return "radio";
          case "range":
            return "slider";
          case "number":
            return "spinbutton";
          case "search":
            return element.getAttribute("list") !== null ? "combobox" : "searchbox";
          case "color":
          case "date":
          case "datetime-local":
          case "file":
          case "hidden":
          case "month":
          case "password":
          case "time":
          case "week":
            return;
          default:
            return element.getAttribute("list") !== null ? "combobox" : "textbox";
        }
      }
      case "a":
      case "area":
        return element.getAttribute("href") !== null ? "link" : undefined;
      case "select":
        return element.getAttribute("multiple") !== null || Number(element.getAttribute("size")) > 1 ? "listbox" : "combobox";
      default:
        return /^h[1-6]$/.test(tag) ? "heading" : Object.hasOwn(NATIVE_ROLES, tag) ? NATIVE_ROLES[tag] : undefined;
    }
  }
  function nameFor(element, root, text) {
    function label(node, referenced) {
      if (!referenced) {
        const ids = normalize(node.getAttribute("aria-labelledby"));
        if (ids) {
          const resolved = ids.split(" ").map((id) => root.getElementById(id));
          const names = resolved.map((target) => target && label(target, true)).filter(Boolean).join(" ");
          if (names)
            return names;
        }
      }
      const aria = normalize(node.getAttribute("aria-label"));
      if (aria)
        return aria;
      if (node.labels?.length) {
        const names = Array.from(node.labels, (target) => normalize(text(target, false))).filter(Boolean).join(" ");
        if (names)
          return names;
      }
      const tag = node.localName;
      const type = node.getAttribute("type")?.toLowerCase();
      if (tag === "img" || tag === "area" || tag === "input" && type === "image") {
        const alt = normalize(node.getAttribute("alt"));
        if (alt)
          return alt;
      }
      if (tag === "input" && (type === "button" || type === "submit" || type === "reset")) {
        const value = normalize(node.getAttribute("value"));
        if (value)
          return value;
        if (type === "submit")
          return "Submit";
        if (type === "reset")
          return "Reset";
      }
      const role = roleFor(node);
      if (referenced || role && Object.hasOwn(CONTENT_ROLES, role)) {
        const contents = normalize(text(node, referenced));
        if (contents)
          return contents;
      }
      return normalize(node.getAttribute("title")) || normalize(node.getAttribute("placeholder"));
    }
    return label(element, false) || undefined;
  }
  function selectorFor(element, root, escape) {
    const unique = (selector, target) => {
      const matches = root.querySelectorAll(selector);
      return matches.length === 1 && matches[0] === target;
    };
    let tail = "";
    let fallback;
    for (let node = element;node; node = node.parentElement) {
      let selector;
      if (node.id) {
        const id = `#${escape(node.id)}`;
        if (unique(id, node))
          selector = id;
      }
      const tag = escape(node.localName);
      if (!selector) {
        for (const attr of SELECTOR_ATTRIBUTES) {
          const value = node.getAttribute(attr);
          if (!value)
            continue;
          const byAttribute = `${tag}[${attr}="${escape(value)}"]`;
          if (unique(byAttribute, node)) {
            selector = byAttribute;
            break;
          }
        }
      }
      if (selector)
        return selector + tail;
      let index = 1;
      let repeated = false;
      for (let sibling = node.previousElementSibling;sibling; sibling = sibling.previousElementSibling) {
        ++index;
        repeated ||= sibling.localName === node.localName;
      }
      for (let sibling = node.nextElementSibling;sibling && !repeated; sibling = sibling.nextElementSibling) {
        repeated = sibling.localName === node.localName;
      }
      const part = repeated ? `${tag}:nth-child(${index})` : tag;
      if (fallback === undefined && unique(part + tail, element))
        fallback = part + tail;
      tail = ` > ${part}${tail}`;
    }
    return fallback;
  }

  // extension/teach/recorder.ts
  var ACTIONABLE = "button, a[href], input, label, select, textarea, summary, [role], [contenteditable]";
  var TYPING_IDLE_MS = 450;
  function visibleText(element) {
    const style = getComputedStyle(element);
    if (style.display === "none" || style.visibility !== "visible" || style.opacity === "0")
      return "";
    if (style.display !== "contents" && element.getClientRects().length === 0)
      return "";
    return element instanceof HTMLElement ? element.innerText : element.textContent ?? "";
  }
  function targetFor(event) {
    let first;
    let actionable;
    for (const item of event.composedPath()) {
      if (!(item instanceof Element))
        continue;
      if (item.localName === OVERLAY_HOST_TAG)
        return;
      first ??= item;
      if (!actionable && item.matches(ACTIONABLE))
        actionable = item;
    }
    return actionable ?? first;
  }
  function describe(kind, element) {
    const root = element.getRootNode();
    const scope = root instanceof ShadowRoot ? root : document;
    return {
      kind,
      url: location.href,
      title: document.title,
      role: roleFor(element),
      name: nameFor(element, scope, (target, includeHidden) => {
        if (includeHidden)
          return target.textContent ?? "";
        return target instanceof Element ? visibleText(target) : "";
      }),
      selector: selectorFor(element, scope, CSS.escape),
      text: (visibleText(element).trim() || (element.parentElement ? visibleText(element.parentElement) : "")).replace(/\s+/g, " ").trim().slice(0, 120) || undefined,
      viewport: { width: innerWidth, height: innerHeight },
      at: Date.now()
    };
  }
  function privateInput(element) {
    return element.type === "password" || element.type === "hidden" || /\b(?:current|new)-password\b/.test(element.autocomplete);
  }
  function typingValue(element) {
    if (element instanceof HTMLInputElement) {
      if (privateInput(element))
        return;
      switch (element.type) {
        case "text":
        case "search":
        case "email":
        case "url":
        case "tel":
        case "number":
          return element.value;
        default:
          return;
      }
    }
    if (element instanceof HTMLTextAreaElement)
      return element.value;
    if (element instanceof HTMLElement && element.isContentEditable)
      return element.innerText;
    return;
  }
  function selectionValue(element) {
    if (element instanceof HTMLSelectElement) {
      return element.multiple ? Array.from(element.selectedOptions, (option) => option.value).join(`
`) : element.value;
    }
    if (element instanceof HTMLInputElement) {
      if (privateInput(element))
        return;
      switch (element.type) {
        case "checkbox":
        case "radio":
          return element.checked ? element.value : "";
        case "date":
        case "datetime-local":
        case "month":
        case "week":
        case "time":
        case "color":
        case "range":
          return element.value;
      }
    }
    return;
  }
  function install() {
    let listeners;
    let pending;
    const delivery = new Set;
    function send(step) {
      const message = { omp: "teach-step", step };
      const sent = chrome.runtime.sendMessage(message).catch(() => {
        return;
      });
      delivery.add(sent);
      sent.then(() => delivery.delete(sent));
    }
    function flush() {
      if (!pending)
        return;
      const { timer, element, step } = pending;
      pending = undefined;
      clearTimeout(timer);
      if (!(element instanceof HTMLInputElement) || !privateInput(element))
        send(step);
    }
    function type(event) {
      if (!event.isTrusted || event instanceof InputEvent && event.isComposing)
        return;
      const element = targetFor(event);
      if (!element)
        return;
      const value = typingValue(element);
      if (value === undefined)
        return;
      if (pending && pending.element !== element)
        flush();
      if (pending) {
        clearTimeout(pending.timer);
        pending.step.value = value;
        pending.step.at = Date.now();
        pending.timer = window.setTimeout(flush, TYPING_IDLE_MS);
      } else {
        const step = describe("type", element);
        step.value = value;
        pending = { element, step, timer: window.setTimeout(flush, TYPING_IDLE_MS) };
      }
    }
    function click(event) {
      if (!event.isTrusted)
        return;
      const target = targetFor(event);
      if (!target)
        return;
      flush();
      send({ ...describe("click", target), x: event.clientX, y: event.clientY });
    }
    function change(event) {
      if (!event.isTrusted)
        return;
      const target = targetFor(event);
      if (!target)
        return;
      flush();
      const value = selectionValue(target);
      if (value !== undefined)
        send({ ...describe("select", target), value });
    }
    function key(event) {
      if (!event.isTrusted || event.isComposing || event.repeat)
        return;
      if (event.key !== "Enter" && event.key !== "Escape" && event.key !== "Tab")
        return;
      const target = targetFor(event);
      if (!target)
        return;
      flush();
      send({ ...describe("key", target), key: event.key });
    }
    function submit(event) {
      if (!event.isTrusted || !(event.target instanceof HTMLFormElement))
        return;
      flush();
      if (!event.submitter)
        send({ ...describe("key", event.target), key: "Enter" });
    }
    chrome.runtime.onMessage.addListener((message, _sender, respond) => {
      if (message?.omp !== "teach")
        return false;
      if (message.op === "start") {
        if (!listeners) {
          listeners = new AbortController;
          const options = { capture: true, signal: listeners.signal };
          document.addEventListener("click", click, options);
          document.addEventListener("input", type, options);
          document.addEventListener("compositionend", type, options);
          document.addEventListener("change", change, options);
          document.addEventListener("keydown", key, options);
          document.addEventListener("submit", submit, options);
          window.addEventListener("pagehide", flush, options);
        }
        respond(true);
        return false;
      }
      if (message.op === "stop") {
        listeners?.abort();
        listeners = undefined;
        flush();
        Promise.allSettled(delivery).then(() => respond(true));
        return true;
      }
      return false;
    });
  }
  if (!globalThis.__ompTeachRecorder) {
    globalThis.__ompTeachRecorder = true;
    install();
  }
})();
