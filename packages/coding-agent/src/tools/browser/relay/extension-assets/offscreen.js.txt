            }
          } else {
            first = Boolean(opts.first);
          }
          width = Math.max(0, Math.floor(width));
          height = Math.max(0, Math.floor(height));
          if (first) {
            if (!palette) {
              throw new Error("First frame must include a { palette } option");
            }
            encodeLogicalScreenDescriptor(stream, width, height, palette, colorDepth);
            encodeColorTable(stream, palette);
            if (repeat >= 0) {
              encodeNetscapeExt(stream, repeat);
            }
          }
          const delayTime = Math.round(delay / 10);
          encodeGraphicControlExt(stream, dispose, delayTime, transparent, transparentIndex);
          const useLocalColorTable = Boolean(palette) && !first;
          encodeImageDescriptor(stream, width, height, useLocalColorTable ? palette : null);
          if (useLocalColorTable)
            encodeColorTable(stream, palette);
          encodePixels(stream, index, width, height, colorDepth, accum, htab, codetab);
        }
      };
      function writeHeader() {
        writeUTFBytes(stream, "GIF89a");
      }
    }
    function encodeGraphicControlExt(stream, dispose, delay, transparent, transparentIndex) {
      stream.writeByte(33);
      stream.writeByte(249);
      stream.writeByte(4);
      if (transparentIndex < 0) {
        transparentIndex = 0;
        transparent = false;
      }
      var transp, disp;
      if (!transparent) {
        transp = 0;
        disp = 0;
      } else {
        transp = 1;
        disp = 2;
      }
      if (dispose >= 0) {
        disp = dispose & 7;
      }
      disp <<= 2;
      const userInput = 0;
      stream.writeByte(0 | disp | userInput | transp);
      writeUInt16(stream, delay);
      stream.writeByte(transparentIndex || 0);
      stream.writeByte(0);
    }
    function encodeLogicalScreenDescriptor(stream, width, height, palette, colorDepth = 8) {
      const globalColorTableFlag = 1;
      const sortFlag = 0;
      const globalColorTableSize = colorTableSize(palette.length) - 1;
      const fields = globalColorTableFlag << 7 | colorDepth - 1 << 4 | sortFlag << 3 | globalColorTableSize;
      const backgroundColorIndex = 0;
      const pixelAspectRatio = 0;
      writeUInt16(stream, width);
      writeUInt16(stream, height);
      stream.writeBytes([fields, backgroundColorIndex, pixelAspectRatio]);
    }
    function encodeNetscapeExt(stream, repeat) {
      stream.writeByte(33);
      stream.writeByte(255);
      stream.writeByte(11);
      writeUTFBytes(stream, "NETSCAPE2.0");
      stream.writeByte(3);
      stream.writeByte(1);
      writeUInt16(stream, repeat);
      stream.writeByte(0);
    }
    function encodeColorTable(stream, palette) {
      const colorTableLength = 1 << colorTableSize(palette.length);
      for (let i = 0;i < colorTableLength; i++) {
        let color = [0, 0, 0];
        if (i < palette.length) {
          color = palette[i];
        }
        stream.writeByte(color[0]);
        stream.writeByte(color[1]);
        stream.writeByte(color[2]);
      }
    }
    function encodeImageDescriptor(stream, width, height, localPalette) {
      stream.writeByte(44);
      writeUInt16(stream, 0);
      writeUInt16(stream, 0);
      writeUInt16(stream, width);
      writeUInt16(stream, height);
      if (localPalette) {
        const interlace = 0;
        const sorted = 0;
        const palSize = colorTableSize(localPalette.length) - 1;
        stream.writeByte(128 | interlace | sorted | 0 | palSize);
      } else {
        stream.writeByte(0);
      }
    }
    function encodePixels(stream, index, width, height, colorDepth = 8, accum, htab, codetab) {
      lzwEncode_default(width, height, index, colorDepth, stream, accum, htab, codetab);
    }
    function writeUInt16(stream, short) {
      stream.writeByte(short & 255);
      stream.writeByte(short >> 8 & 255);
    }
    function writeUTFBytes(stream, text) {
      for (var i = 0;i < text.length; i++) {
        stream.writeByte(text.charCodeAt(i));
      }
    }
    function colorTableSize(length) {
      return Math.max(Math.ceil(Math.log2(length)), 1);
    }
    var src_default = GIFEncoder;
  });

  // extension/offscreen/layout.ts
  function frameLayout(canvas, frame, index, count) {
    const scale = canvas.width / frame.viewport.width;
    if (!Number.isFinite(scale) || scale <= 0)
      throw new Error("GIF frame has no viewport width");
    const width = frame.viewport.width;
    const height = canvas.height / scale;
    const markSize = 28;
    const action = frame.action?.kind;
    return {
      scale,
      width,
      height,
      mark: { x: width - markSize - 8, y: height - markSize - 12, size: markSize },
      progress: { x: 0, y: height - 3, width: width * ((index + 1) / count), height: 3 },
      delay: (frame.delayMs ?? (action === "click" || action === "drag" ? 1500 : 800)) + (index + 1 === count ? 2000 : 0)
    };
  }
  function labelBox(viewport, labelWidth, anchor) {
    const inset = 8;
    const height = 28;
    const maxX = Math.max(0, viewport.width - labelWidth - inset);
    const maxY = Math.max(0, viewport.height - height - inset);
    if (!anchor)
      return { x: Math.min(inset, maxX), y: maxY, height };
    const right = anchor.x + 20;
    const x = right <= maxX ? right : anchor.x - 20 - labelWidth;
    return { x: Math.max(0, Math.min(x, maxX)), y: Math.max(0, Math.min(anchor.y - 10, maxY)), height };
  }
  function dragHead(from, to) {
    const dx = to.x - from.x;
    const dy = to.y - from.y;
    const distance = Math.hypot(dx, dy);
    const length = Math.min(14, distance - 12);
    if (length <= 0)
      return;
    const ux = dx / distance;
    const uy = dy / distance;
    const tip = { x: to.x - 6 * ux, y: to.y - 6 * uy };
    const base = { x: tip.x - length * ux, y: tip.y - length * uy };
    const halfWidth = length / 2;
    return {
      tip,
      left: { x: base.x - uy * halfWidth, y: base.y + ux * halfWidth },
      right: { x: base.x + uy * halfWidth, y: base.y - ux * halfWidth }
    };
  }

  // extension/offscreen/main.ts
  var CYAN = "#5ad8e6";
  var PINK = "#ed4abf";
  var OMP_MARK = new Path2D("M10 14h44v9H43v33h-9V23h-9v22h-9V23H10z");
  function drawClick(ctx, point) {
    ctx.lineWidth = 2;
    ctx.strokeStyle = "rgba(90,216,230,.5)";
    ctx.beginPath();
    ctx.arc(point.x, point.y, 11, 0, 2 * Math.PI);
    ctx.stroke();
    ctx.strokeStyle = "rgba(90,216,230,.3)";
    ctx.beginPath();
    ctx.arc(point.x, point.y, 15, 0, 2 * Math.PI);
    ctx.stroke();
  }
  function drawDrag(ctx, from, to) {
    ctx.strokeStyle = PINK;
    ctx.lineWidth = 3;
    ctx.beginPath();
    ctx.moveTo(from.x, from.y);
    ctx.lineTo(to.x, to.y);
    ctx.stroke();
    const head = dragHead(from, to);
    if (head) {
      ctx.fillStyle = PINK;
      ctx.beginPath();
      ctx.moveTo(head.tip.x, head.tip.y);
      ctx.lineTo(head.left.x, head.left.y);
      ctx.lineTo(head.right.x, head.right.y);
      ctx.closePath();
      ctx.fill();
    }
    ctx.fillStyle = "#ededef";
    ctx.lineWidth = 2;
    for (const point of [from, to]) {
      ctx.beginPath();
      ctx.arc(point.x, point.y, 6, 0, 2 * Math.PI);
      ctx.fill();
      ctx.stroke();
    }
  }
  function drawLabel(ctx, viewport, label, anchor) {
    const padding = 8;
    ctx.font = "500 14px system-ui, sans-serif";
    ctx.textBaseline = "middle";
    const width = Math.min(ctx.measureText(label).width + 2 * padding, viewport.width - 2 * padding);
    if (width <= 2 * padding)
      return;
    const { x, y, height } = labelBox(viewport, width, anchor);
    ctx.fillStyle = "rgba(10,10,11,.92)";
    ctx.beginPath();
    ctx.roundRect(x, y, width, height, 8);
    ctx.fill();
    ctx.fillStyle = "#ededef";
    ctx.fillText(label, x + padding, y + height / 2, width - 2 * padding);
  }
  function drawAction(ctx, viewport, action) {
    switch (action.kind) {
      case "click":
        drawClick(ctx, action);
        drawLabel(ctx, viewport, action.label || "Clicking", action);
        break;
      case "drag":
        drawDrag(ctx, action.from, action.to);
        drawLabel(ctx, viewport, action.label || "Dragging", action.to);
        break;
      case "scroll":
        drawLabel(ctx, viewport, action.label || "Scrolling", action);
        break;
      case "type":
        drawLabel(ctx, viewport, action.label || "Typing");
        break;
    }
  }
  function drawMark(ctx, { x, y, size }) {
    ctx.save();
    ctx.translate(x, y);
    ctx.scale(size / 64, size / 64);
    ctx.fillStyle = "rgba(10,10,11,.75)";
    ctx.beginPath();
    ctx.roundRect(0, 0, 64, 64, 16);
    ctx.fill();
    const gradient = ctx.createLinearGradient(10, 14, 54, 56);
    gradient.addColorStop(0, "oklch(0.7 0.24 340)");
    gradient.addColorStop(0.5, "oklch(0.62 0.21 295)");
    gradient.addColorStop(1, "oklch(0.81 0.14 200)");
    ctx.fillStyle = gradient;
    ctx.fill(OMP_MARK);
    ctx.restore();
  }
  async function decode({ image }) {
    const encoded = await fetch(`data:${image.mimeType};base64,${image.data}`);
    return createImageBitmap(await encoded.blob());
  }
  async function encode(frames) {
    const first = frames[0];
    if (!first)
      throw new Error("There are no screenshots to export");
    const encoder = import_gifenc.GIFEncoder();
    let firstImage = await decode(first);
    try {
      const canvas = document.createElement("canvas");
      canvas.width = firstImage.width;
      canvas.height = firstImage.height;
      const ctx = canvas.getContext("2d", { alpha: false, willReadFrequently: true });
      if (!ctx)
        throw new Error("Chrome could not allocate a GIF canvas");
      for (const [index, frame] of frames.entries()) {
        const layout = frameLayout(canvas, frame, index, frames.length);
        const image = firstImage ?? await decode(frame);
        firstImage = undefined;
        try {
          ctx.resetTransform();
          ctx.fillStyle = "#09090a";
          ctx.fillRect(0, 0, canvas.width, canvas.height);
          ctx.drawImage(image, 0, 0, canvas.width, canvas.width * image.height / image.width);
        } finally {
          image.close();
        }
        ctx.setTransform(layout.scale, 0, 0, layout.scale, 0, 0);
        if (frame.action)
          drawAction(ctx, layout, frame.action);
        drawMark(ctx, layout.mark);
        const gradient = ctx.createLinearGradient(0, 0, layout.width, 0);
        gradient.addColorStop(0, CYAN);
        gradient.addColorStop(1, PINK);
        ctx.fillStyle = gradient;
        const bar = layout.progress;
        ctx.fillRect(bar.x, bar.y, bar.width, bar.height);
        const { data } = ctx.getImageData(0, 0, canvas.width, canvas.height);
        const palette = import_gifenc.quantize(data, 256);
        encoder.writeFrame(import_gifenc.applyPalette(data, palette), canvas.width, canvas.height, { palette, delay: layout.delay });
      }
      encoder.finish();
    } finally {
      firstImage?.close();
    }
    const blob = new Blob([encoder.bytesView()], { type: "image/gif" });
    const reader = new FileReader;
    const result = Promise.withResolvers();
    reader.addEventListener("load", () => {
      if (typeof reader.result === "string")
        result.resolve(reader.result);
      else
        result.reject(new Error("Could not read the GIF data URL"));
    }, { once: true });
    reader.addEventListener("error", () => result.reject(reader.error ?? new Error("Could not read the GIF")), { once: true });
    reader.readAsDataURL(blob);
    return result.promise;
  }
  chrome.runtime.onMessage.addListener((message, _sender, sendResponse) => {
    if (message?.omp !== "encode-gif")
      return false;
    encode(message.frames).then((url) => sendResponse({ ok: true, url }), (error) => sendResponse({ ok: false, error: error instanceof Error ? error.message : String(error) }));
    return true;
  });
