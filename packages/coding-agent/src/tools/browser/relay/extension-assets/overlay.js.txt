  var EDGE_GAP = 8;
  function placeChip(cursor, width, viewport) {
    const bottom = cursor.y + 38 > viewport.height - EDGE_GAP;
    const flipped = cursor.x + (bottom ? 24 : 13) + width > viewport.width - EDGE_GAP;
    const wanted = bottom ? flipped ? -6 - width : 24 : flipped ? 8 - width : 13;
    const dy = bottom ? Math.min(2, viewport.height - 30 - cursor.y) : flipped ? 18 : 16;
    const dx = Math.max(EDGE_GAP - cursor.x, Math.min(viewport.width - EDGE_GAP - width - cursor.x, wanted));
    return { dx, dy, flipped, bottom };
  }
  var NO_GESTURE = { pressedAt: null, dragging: false };
  var DRAG_SLOP = 3;
  function nameInput(gesture, input) {
    switch (input.kind) {
      case "press":
        return { gesture: { pressedAt: { x: input.x, y: input.y }, dragging: false }, verb: "Clicking" };
      case "move": {
        const { pressedAt } = gesture;
        if (!pressedAt)
          return { gesture, verb: "Moving" };
        const dragging = gesture.dragging || Math.hypot(input.x - pressedAt.x, input.y - pressedAt.y) > DRAG_SLOP;
        return { gesture: { pressedAt, dragging }, verb: dragging ? "Dragging" : "Clicking" };
      }
      case "release":
        return { gesture: NO_GESTURE, verb: gesture.dragging ? "Dragging" : "Clicking" };
      case "wheel":
        return { gesture, verb: "Scrolling" };
      case "key":
        return { gesture, verb: "Typing" };
    }
  }

  // extension/overlay/motion.ts
  var TRIP = {
    baseMs: 80,
    perBitMs: 70,
    targetPx: 24,
    minMs: 140,
    maxMs: 380,
    snapPx: 0.5,
    arc: 0.1,
    arcMaxPx: 30,
    elbow: { x: 0.6, y: 0.8 },
    overshoot: 0.03,
    overshootMaxPx: 4,
    mainEnds: 0.8,
    fixStarts: 0.66,
    steps: 16,
    leanDeg: 9,
    leanAtPxPerMs: 1.8
  };
  function minimumJerk(s) {
    if (s <= 0)
      return 0;
    if (s >= 1)
      return 1;
    return s * s * s * (10 - 15 * s + 6 * s * s);
  }
  function tripDuration(distance) {
    if (distance < TRIP.snapPx)
      return 0;
    const ms = TRIP.baseMs + TRIP.perBitMs * Math.log2(1 + distance / TRIP.targetPx);
    return Math.round(Math.max(TRIP.minMs, Math.min(TRIP.maxMs, ms)));
  }
  function planTrip(from, to) {
    const dx = to.x - from.x;
    const dy = to.y - from.y;
    const distance = Math.hypot(dx, dy);
    const durationMs = tripDuration(distance);
    if (durationMs === 0)
      return { durationMs, points: [{ x: to.x, y: to.y }], leanDeg: [0] };
    const overshoot = Math.min(TRIP.overshoot * distance, TRIP.overshootMaxPx);
    const side = dx * TRIP.elbow.y - dy * TRIP.elbow.x > 0 ? -1 : 1;
    const arc = side * Math.min(TRIP.arc * distance, TRIP.arcMaxPx);
    const points = [];
    for (let i = 0;i <= TRIP.steps; i++) {
      const s = i / TRIP.steps;
      const correction = overshoot * minimumJerk((s - TRIP.fixStarts) / (1 - TRIP.fixStarts));
      const along = ((distance + overshoot) * minimumJerk(s / TRIP.mainEnds) - correction) / distance;
      const across = arc * Math.sin(Math.PI * Math.max(0, Math.min(1, along))) / distance;
      points.push({ x: from.x + dx * along - dy * across, y: from.y + dy * along + dx * across });
    }
    points[TRIP.steps] = { x: to.x, y: to.y };
    return { durationMs, points, leanDeg: leanAngles(points, durationMs) };
  }
  function tripPointAt(trip, elapsedMs) {
    const last = trip.points.length - 1;
    if (trip.durationMs === 0 || elapsedMs >= trip.durationMs)
      return trip.points[last];
    const position = Math.max(0, elapsedMs) / trip.durationMs * last;
    const index = Math.floor(position);
    const t = position - index;
    const a = trip.points[index];
    const b = trip.points[index + 1];
    return { x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t };
  }
  function leanAngles(points, durationMs) {
    const last = points.length - 1;
    const stepMs = durationMs / last;
    return points.map((_, i) => {
      if (i === 0 || i === last)
        return 0;
      const before = points[i - 1];
      const after = points[i + 1];
      const speed = (after.x - before.x - (after.y - before.y)) * Math.SQRT1_2 / (2 * stepMs);
      return TRIP.leanDeg * Math.tanh(speed / TRIP.leanAtPxPerMs);
    });
  }

        this.#shown = true;

      }

    }
    #rest() {
      if (this.#gesture.pressedAt && performance.now() - this.#lastInputAt < HELD_LIMIT_MS) {
        this.#restTimer = window.setTimeout(() => this.#rest(), REST_AFTER_MS);
        return;
      }
      this.#gesture = NO_GESTURE;
      this.#chip.classList.remove("on");
      this.#cursor.classList.add("rest");

    }
    #tip() {
      if (!this.#trip)
        return this.#at;
      const elapsed = this.#trip.run.currentTime;
      return tripPointAt(this.#trip.plan, typeof elapsed === "number" ? elapsed : 0);
    }
    #travel(to) {
      if (Math.hypot(to.x - this.#at.x, to.y - this.#at.y) < TRIP.snapPx)
        return;
      const from = this.#tip();
      this.#land();
      this.#at = to;
      this.#cursor.style.setProperty("transform", translate(to));
      const plan = planTrip(from, to);
      if (plan.durationMs === 0 || skipMotion())
        return;
      const timing = { duration: plan.durationMs, easing: "linear" };
      this.#leaning = this.#lean.animate(plan.leanDeg.map((degrees) => ({ transform: `rotate(${degrees.toFixed(2)}deg)` })), timing);
      const run = this.#cursor.animate(plan.points.map((point) => ({ transform: translate(point) })), timing);
      this.#trip = { plan, run };
      run.onfinish = () => {
        if (this.#trip?.run === run)
          this.#land();
      };
    }
    #land() {
      this.#trip?.run.cancel();
      this.#leaning?.cancel();
      this.#trip = null;
      this.#leaning = null;
      const queued = this.#queuedRing;
      this.#queuedRing = null;
      if (queued)
        this.#ring(queued);
    }
    #ring(point) {
      const ring = el("div", "ring");
      ring.style.setProperty("translate", `${point.x.toFixed(2)}px ${point.y.toFixed(2)}px`);
      const remove = () => ring.remove();
      ring.addEventListener("animationend", remove);
      ring.addEventListener("animationcancel", remove);
      this.layer.append(ring);
    }
    #label(verb) {
      const chip = this.#chip;
      if (chip.textContent !== verb)
        chip.textContent = verb;
      let width = this.#chipWidths.get(verb);
      if (width === undefined) {
        width = chip.offsetWidth;
        if (width > 0)
          this.#chipWidths.set(verb, width);
      }
      const placement = placeChip(this.#at, width, { width: window.innerWidth, height: window.innerHeight });
      chip.style.setProperty("transform", `translate(${placement.dx}px, ${placement.dy}px)`);
      chip.classList.toggle("flip", placement.flipped);
      chip.classList.add("on");
    }
  }

  class Presence {
    layers;
    #emit;

    #frame = el("div", "frame");

    constructor(emit) {
      this.#emit = emit;

      this.#move.addEventListener("click", () => {
        this.#preferred = this.#edge === "top" ? "bottom" : "top";
        this.#dodging = false;
        this.#syncEdge();
      });

      this.#ask.setAttribute("role", "group");
      this.#ask.setAttribute("aria-label", "omp");
      this.#ask.append(dot, "omp needs you", this.#openButton());

      open.addEventListener("click", () => this.#emit("open"));
      const dismiss = button("", "Dismiss", icon(svg("path", { d: "M4 4l8 8M12 4l-8 8" })), el("span", "tip", "Dismiss"));
      dismiss.addEventListener("click", () => {
        if (this.#activity.state !== "idle")
          this.#dismissed = this.#activity.session;
        this.#emit("dismiss");
        this.#render();
      });
      const rule = el("span", "rule");
      rule.setAttribute("aria-hidden", "true");
      this.#pill.setAttribute("role", "group");
      this.#pill.setAttribute("aria-label", "omp");
      this.#pill.append(this.#pillLead, this.#pillText, rule, open, dismiss);
      this.#toast.setAttribute("role", "status");
      this.#syncEdgeLabels();
      this.layers = [this.#glow, this.#tag, this.#ask, this.#pill, this.#toast, this.#frame];
    }
    setActivity(activity, role) {
      const wasWorking = this.#activity.state === "working";
      this.#activity = activity;
      this.#role = role;
      if (activity.state === "idle")
        this.#dismissed = null;
      if (activity.state !== "working") {
        this.#dodging = false;
        if (wasWorking)
          this.#endStop();
      }
      this.#render();
    }
    setEdgeEnabled(enabled) {
      this.#edgeEnabled = enabled;
      this.#render();
    }
    aim(point) {
      const dodging = point !== null && this.#covers(point);
      if (dodging === this.#dodging)
        return;
      this.#dodging = dodging;
      this.#syncEdge();
    }
    occupied() {
      const reach = { top: 0, bottom: 0 };
      if (this.#tag.classList.contains("on"))
        reach[this.#edge] = this.#tag.offsetHeight;
      if (this.#ask.classList.contains("on"))
        reach[this.#edge] = 20 + this.#ask.offsetHeight;
      if (this.#pill.classList.contains("on"))
        reach.bottom = 16 + this.#pill.offsetHeight;
      return reach;
    }
    stopAck(delivered) {
      if (!this.#stopping)
        return;
      clearTimeout(this.#stopTimer);
      if (delivered)
        this.#stopTimer = window.setTimeout(() => this.#endStop(), STOP_RETRY_MS);
      else
        this.#failStop();
    }
    #covers(point) {
      if (this.#role !== "main" || this.#activity.state !== "working")
        return false;
      const { width, height } = this.#tag.getBoundingClientRect();
      const left = (window.innerWidth - width) / 2;
      if (point.x < left - 96 || point.x > left + width + 32)
        return false;
      return this.#preferred === "top" ? point.y < height + 12 : point.y > window.innerHeight - height - 44;
    }
    #openButton() {
      const open = button("open", "Open omp", "Open omp");
      open.addEventListener("click", () => this.#emit("open"));
      return open;
    }
    #render() {
      const activity = this.#activity;
      const main = this.#role === "main";
      const working = activity.state === "working";
      const lit = this.#edgeEnabled && working;
      this.#glow.classList.toggle("on", lit);
      this.#frame.classList.toggle("on", lit && main);
      const tagWasOn = this.#tag.classList.contains("on");
      this.#tag.classList.toggle("on", main && working);
      this.#ask.classList.toggle("on", main && activity.state === "waiting");
      this.#stop.hidden = !(working && activity.canStop);
      this.#setLabel(main && working ? activity.label : undefined, tagWasOn && main && working);
      const pill = !main && activity.state !== "idle" && activity.session !== this.#dismissed;
      this.#pill.classList.toggle("on", pill);
      if (pill) {
        const lead = working ? ompMark() : el("span", "dot");
        lead.setAttribute("aria-hidden", "true");
        this.#pillLead.classList.toggle("on", !working);
        this.#pillLead.replaceChildren(lead);
        this.#pillText.textContent = working ? "omp is working in this group" : "omp needs you in this group";
      }
    }
    #syncEdge() {
      const other = this.#preferred === "top" ? "bottom" : "top";
      const edge = this.#dodging ? other : this.#preferred;
      if (edge === this.#edge)
        return;
      const from = this.#edge;
      this.#edge = edge;
      if (this.#edgeMoving)
        return;
      const tag = this.#tag;
      if (!tag.classList.contains("on") || skipMotion()) {
        this.#syncEdgeLabels();
        return;
      }
      const away = (side) => ({ transform: `translateY(${side === "top" ? -100 : 100}%)`, opacity: 0 });
      const home = { transform: "translateY(0)", opacity: 1 };
      this.#edgeMoving = true;
      const leave = tag.animate([home, away(from)], { duration: TAG_LEAVE_MS, easing: "ease-in", fill: "forwards" });
      leave.onfinish = () => {
        this.#edgeMoving = false;
        this.#syncEdgeLabels();
        leave.cancel();
        tag.animate([away(this.#edge), home], { duration: TAG_ENTER_MS, easing: "cubic-bezier(.16,1,.3,1)" });
      };
    }
    #syncEdgeLabels() {
      this.#tag.dataset.edge = this.#edge;
      this.#ask.dataset.edge = this.#edge;
      const label = this.#edge === "top" ? "Move to bottom" : "Move to top";
      this.#move.setAttribute("aria-label", label);
      this.#moveTip.textContent = label;
    }
    #setLabel(label, animate) {
      if (label === this.#label)
        return;
      this.#label = label;
      const swap = ++this.#labelSwap;
      const box = this.#live;
      const text = this.#liveText;
      if (!animate || skipMotion()) {
        text.textContent = label ?? "";
        return;
      }
      const from = box.getBoundingClientRect().width;
      text.animate([{ opacity: 1 }, { opacity: 0, transform: "translateY(-3px)" }], {
        duration: LABEL_EXIT_MS,
        easing: "ease-out",
        fill: "forwards"
      });
      setTimeout(() => {
        if (swap !== this.#labelSwap)
          return;
        for (const animation of text.getAnimations())
          animation.cancel();
        text.textContent = label ?? "";
        const to = box.getBoundingClientRect().width;
        const timing = { duration: LABEL_ENTER_MS, easing: "ease-in-out" };
        box.animate([{ width: `${from}px` }, { width: `${to}px` }], timing);
        text.animate([{ opacity: 0, transform: "translateY(3px)" }, { opacity: 1 }], timing);
      }, LABEL_ENTER_DELAY_MS);
    }
    #pressStop() {
      if (this.#stopping)
        return;
      this.#stopping = true;
      this.#stop.disabled = true;
      this.#stopText.textContent = "Stopping…";
      this.#emit("stop");
      this.#stopTimer = window.setTimeout(() => this.#failStop(), STOP_ACK_MS);
    }
    #endStop() {
      clearTimeout(this.#stopTimer);
      this.#stopping = false;
      this.#stop.disabled = false;
      this.#stopText.textContent = "Stop";
    }
    #failStop() {
      this.#endStop();
      this.#toast.textContent = STOP_FAILED;
      this.#toast.classList.add("on");
      clearTimeout(this.#toastTimer);
      this.#toastTimer = window.setTimeout(() => this.#toast.classList.remove("on"), TOAST_MS);
    }
  }

  // extension/overlay/overlay.ts
  var CONCEAL_LIMIT_MS = 15000;
  function readSwitches(settings) {
    const stored = typeof settings === "object" && settings !== null && "overlay" in settings ? settings.overlay : null;
    const overlay = typeof stored === "object" && stored !== null ? stored : {};
    return {
      enabled: !(("enabled" in overlay) && overlay.enabled === false),
      arrow: !(("arrow" in overlay) && overlay.arrow === false),
      edge: !(("edge" in overlay) && overlay.edge === false)
    };
  }

  class Overlay {
    #mounted = null;
    #switches = { enabled: true, arrow: true, edge: true };
    #activity = { state: "idle" };
    #role = "main";
    #conceals = 0;
    #concealTimer;
    #storageListener = (changes, area) => {
      if (area === "local" && SETTINGS_KEY in changes)
        this.#applySwitches(readSwitches(changes[SETTINGS_KEY].newValue));
    };
    constructor() {
      chrome.storage.onChanged.addListener(this.#storageListener);
      chrome.storage.local.get(SETTINGS_KEY).then((stored) => this.#applySwitches(readSwitches(stored[SETTINGS_KEY])), () => {});
    }
    async handle(command) {
      switch (command.op) {
        case "mount":
          this.#mount();
          return true;
        case "unmount":
          this.#unmount();
          return true;
        case "input":
          if (this.#switches.enabled && this.#switches.arrow)
            this.#mount().pointer.input(command.input);
          return true;
        case "activity":
          this.#activity = command.activity;
          this.#role = command.role;
          if (command.activity.state !== "idle" || this.#mounted)
            this.#showActivity(this.#mount());
          return true;
        case "hide":
          this.#conceals++;
          clearTimeout(this.#concealTimer);
          this.#concealTimer = window.setTimeout(() => {
            this.#conceals = 0;
            this.#syncConcealed();
          }, CONCEAL_LIMIT_MS);
          if (!this.#mounted)
            return true;
          this.#syncConcealed();
          await afterFrames(2);
          return true;
        case "show":
          this.#conceals = Math.max(0, this.#conceals - 1);
          if (this.#conceals === 0)
            clearTimeout(this.#concealTimer);
          this.#syncConcealed();
          return true;
        case "pick":
          return await this.#mount().picker.pick(command.mode);
        case "cancelPick":
          this.#mounted?.picker.cancel();
          return true;
        case "stopAck":
          this.#mounted?.presence.stopAck(command.delivered);
          return true;
      }
    }
    dispose() {
      this.#unmount();
      try {
        chrome.storage.onChanged.removeListener(this.#storageListener);
      } catch {}
    }
    #mount() {
      if (this.#mounted)
        return this.#mounted;
      for (const stale of document.querySelectorAll(OVERLAY_HOST_TAG))
        stale.remove();
      const host = document.createElement(OVERLAY_HOST_TAG);
      const shadow = host.attachShadow({ mode: "closed" });
      const sheet = new CSSStyleSheet;
      sheet.replaceSync(OVERLAY_CSS);
      shadow.adoptedStyleSheets = [sheet];
      const presence = new Presence((op) => {
        const event = { omp: "overlay-event", op };
        try {
          chrome.runtime.sendMessage(event).catch(() => {});
        } catch {}
      });
      const pointer = new Pointer((point) => presence.aim(point));
      const stage = el("div", "stage", ...presence.layers, pointer.layer);
      const pickerLayer = el("div", "picker");
      shadow.append(stage, pickerLayer);
      const listeners = new AbortController;
      const { signal } = listeners;
      window.addEventListener("resize", () => pointer.relayout(), { passive: true, signal });
      window.addEventListener("pagehide", () => this.#quiet(), { signal });
      const mounted = {
        host,
        stage,
        pickerLayer,
        pointer,
        presence,
        picker: new Picker(pickerLayer, host, () => presence.occupied()),
        favicon: new Favicon,
        listeners
      };
      this.#mounted = mounted;
      this.#syncConcealed();
      this.#applySwitches(this.#switches);
      document.documentElement.append(host);
      return mounted;
    }
    #unmount() {
      const mounted = this.#mounted;
      if (!mounted)
        return;
      this.#mounted = null;
      this.#activity = { state: "idle" };
      mounted.listeners.abort();
      mounted.picker.cancel();
      mounted.favicon.dispose();
      mounted.host.remove();
    }
    #quiet() {
      const mounted = this.#mounted;
      if (!mounted)
        return;
      this.#activity = { state: "idle" };
      mounted.picker.cancel();
      this.#showActivity(mounted);
    }
    #showActivity(mounted) {
      const activity = this.#activity;
      const working = activity.state === "working" && this.#role === "main";
      mounted.presence.setActivity(activity, this.#role);
      mounted.favicon.set(working && this.#switches.enabled);
      mounted.pointer.setPersistent(working);
      if (activity.state !== "working")
        mounted.pointer.clear();
    }
    #applySwitches(switches) {
      this.#switches = switches;
      const mounted = this.#mounted;
      if (!mounted)
        return;
      mounted.stage.classList.toggle("off", !switches.enabled);
      mounted.presence.setEdgeEnabled(switches.edge);
      mounted.favicon.set(switches.enabled && this.#role === "main" && this.#activity.state === "working");
      if (!switches.enabled || !switches.arrow)
        mounted.pointer.clear();
    }
    #syncConcealed() {
      const mounted = this.#mounted;
      if (!mounted)
        return;
      const concealed = this.#conceals > 0;
      mounted.stage.classList.toggle("concealed", concealed);
      mounted.pickerLayer.classList.toggle("concealed", concealed);
    }
  }

  // extension/overlay/main.ts
  var LIVENESS_MS = 2000;
  function isOverlayCommand(message) {
    return typeof message === "object" && message !== null && "omp" in message && message.omp === "overlay";
  }
  function alive() {
    try {
      return chrome.runtime.id !== undefined;
    } catch {
      return false;
    }
  }
  function install() {
    const overlay = new Overlay;
    const onMessage = (message, _sender, sendResponse) => {
      if (!isOverlayCommand(message))
        return false;
      overlay.handle(message).then(sendResponse, () => sendResponse(null));
      return true;
    };
    const dispose = () => {
      clearInterval(watch);
      globalThis.ompOverlay = undefined;
      overlay.dispose();
      try {
        chrome.runtime.onMessage.removeListener(onMessage);
      } catch {}
    };
    const watch = setInterval(() => {
      if (!alive())
        dispose();
    }, LIVENESS_MS);
    chrome.runtime.onMessage.addListener(onMessage);
    globalThis.ompOverlay = { alive, dispose };
  }
  if (window.top === window && !globalThis.ompOverlay?.alive()) {
    globalThis.ompOverlay?.dispose();
    install();
  }
})();
