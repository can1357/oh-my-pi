  var RELAY_CALL_TIMEOUT_MS = 15000;

  class RelayCalls {
    #pending = new Map;
    #timeoutMs;
    #nextId = 1;
    constructor(timeoutMs = RELAY_CALL_TIMEOUT_MS) {
      this.#timeoutMs = timeoutMs;
    }
    start(call, send) {
      const id = this.#nextId++;
      const { promise, resolve, reject } = Promise.withResolvers();
      if (!send({ t: "call", id, call })) {
        reject(new Error("omp relay is not connected"));
        return promise;
      }
      const timer = setTimeout(() => {
        this.#pending.delete(id);
        reject(new Error(`omp relay did not answer ${call.op} within ${this.#timeoutMs / 1000}s`));
      }, this.#timeoutMs);
      this.#pending.set(id, { resolve, reject, timer });
      return promise;
    }
    settle(message) {
      const pending = this.#pending.get(message.id);
      if (!pending)
        return;
      this.#pending.delete(message.id);
      clearTimeout(pending.timer);
      if (message.ok)
        pending.resolve(message.result);
      else
        pending.reject(new Error(message.error ?? "omp relay call failed"));
    }
    failAll(reason) {
      for (const pending of this.#pending.values()) {
        clearTimeout(pending.timer);
        pending.reject(new Error(reason));
      }
      this.#pending.clear();
    }
  }

  function ensureInstanceId() {
    instanceId ??= (async () => {
      const stored = await chrome.storage.local.get({ [INSTANCE_KEY]: "" });
      const existing = stored[INSTANCE_KEY];
      if (typeof existing === "string" && existing.length > 0)
        return existing;
      const created = crypto.randomUUID();
      await chrome.storage.local.set({ [INSTANCE_KEY]: created });
      return created;
    })();
    return instanceId;
  }
  var browserSession = null;

    browserSession ??= (async () => {
      const stored = await chrome.storage.session.get({ [BROWSER_SESSION_KEY]: "" });
      const existing = stored[BROWSER_SESSION_KEY];
      if (typeof existing === "string" && existing.length > 0)
        return existing;
      const created = crypto.randomUUID();
      await chrome.storage.session.set({ [BROWSER_SESSION_KEY]: created });
      return created;
    })();

  async function instanceCode(instance, session) {
    const identity = session ? `${instance}/${session}` : instance;
    const digest = new Uint8Array(await crypto.subtle.digest("SHA-256", new TextEncoder().encode(identity)));
    let binary = "";
    for (const byte of digest.subarray(0, 6))
      binary += String.fromCharCode(byte);
    return btoa(binary).replaceAll("+", "-").replaceAll("/", "_");
  }
  async function pageTargetId(tabId) {
    const [instance, session] = await Promise.all([ensureInstanceId(), ensureBrowserSession()]);
    return `PAGE${await instanceCode(instance, session)}.${tabId}`;

      }

  function handleRelayLine(raw, origin, sendReply) {
    let msg;
    try {
      msg = JSON.parse(raw);
    } catch {
      return;
    }
    switch (msg.t) {
      case "reload":
        chrome.runtime.reload();
        return;
      case "input":
        handlers?.input(msg.tabId, msg.input);
        return;
      case "callResult":
        origin.calls.settle(msg);
        return;
      case "rpc": {
        const answer = (reply) => {
          if (origin.open)
            sendReply(reply);
        };
        const { id } = msg;
        const run = handlers ? handlers.rpc(msg, origin) : Promise.reject(new Error("worker not ready"));
        run.then((result) => answer({ t: "rpcResult", id, ok: true, result })).catch((err) => {
          answer({
            t: "rpcResult",
            id,
            ok: false,
            error: err instanceof Error ? err.message : String(err)
          });
        });
        return;
      }
    }
  }

  function connectChatRelay(port) {
    const origin = new Link;
    const send = (message) => {
      if (!origin.open)
        return;
      try {
        port.postMessage({ t: "relay", line: JSON.stringify(message) });
      } catch {
        close();
      }
    };
    const close = () => {
      origin.close("chat relay disconnected");
      chatLinks.delete(send);
    };
    buildHello().then((hello) => {
      if (!origin.open)
        return;
      chatLinks.add(send);
      send(hello);
    }).catch(() => port.disconnect());
    return { receive: (line) => handleRelayLine(line, origin, send), close };
  }
  function linked() {
    if (link)
      unlinked("omp relay changed");
    clearLinkDeadline();
    reconnectDelay = RECONNECT_MIN_MS;
    const epoch = new Link;
    link = epoch;
    setBadge(true);
    buildHello().then((hello) => {
      if (link !== epoch || !post(hello))
        return;
      if (busyReasons.size > 0)
        post({ t: "busy", busy: true });
    });
    handlers?.status(relayStatus());
  }
  function unlinked(reason) {
    clearLinkDeadline();
    const epoch = link;
    link = null;
    setBadge(false);
    if (!epoch)
      return;
    epoch.close(reason);
    handlers?.disconnected();
    handlers?.status(relayStatus());
  }
  function scheduleReconnect() {
    const delay = reconnectDelay;
    reconnectDelay = Math.min(reconnectDelay * 2, RECONNECT_MAX_MS);
    clearTimeout(reconnectTimer ?? undefined);
    reconnectTimer = setTimeout(() => {
      reconnectTimer = null;
      connectRelay();
    }, delay);
  }
  function clearLinkDeadline() {
    clearTimeout(deadlineTimer ?? undefined);
    deadlineTimer = null;
  }
  function enforceLinkDeadline(port) {
    clearLinkDeadline();
    deadlineTimer = setTimeout(() => {
      deadlineTimer = null;
      if (link || hostPort !== port)
        return;
      hostPort = null;
      port.disconnect();
      scheduleReconnect();
    }, LINK_DEADLINE_MS);
  }
  function connectRelay() {
    if (hostPort)
      return;
    const port = chrome.runtime.connectNative(NATIVE_HOST);
    hostPort = port;
    enforceLinkDeadline(port);
    const chunks = new Map;
    port.onMessage.addListener((message) => {
      if (hostPort !== port)
        return;
      switch (message.t) {
        case "linked":
          linked();
          return;
        case "unlinked":
          unlinked(message.reason);
          return;
        case "relay":
          if (link)
            handleRelayLine(message.line, link, post);
          return;
        case "chunk": {
          const parts = chunks.get(message.id) ?? [];
          parts[message.index] = message.data;
          if (!message.last) {
            chunks.set(message.id, parts);
            return;
          }
          chunks.delete(message.id);
          if (link)
            handleRelayLine(parts.join(""), link, post);
          return;
        }
      }
    });
    port.onDisconnect.addListener(() => {
      const detail = chrome.runtime.lastError?.message;
      if (hostPort !== port)
        return;
      hostPort = null;
      clearLinkDeadline();
      unlinked(detail ? `omp native host disconnected: ${detail}` : "omp native host disconnected");
      scheduleReconnect();
    });
    const request = { t: "link", extensionId: chrome.runtime.id };
    port.postMessage(request);
  }

  class SessionValue {
    #key;
    #initial;
    #value = null;
    constructor(key, initial) {
      this.#key = key;
      this.#initial = initial;
    }
    get() {
      this.#value ??= chrome.storage.session.get(this.#key).then((stored) => stored[this.#key] ?? this.#initial()).catch(() => this.#initial());

      await chrome.storage.session.set({ [this.#key]: value }).catch(() => {});
    }
  }

  var OVERLAY_SCRIPT = "overlay.js";
  var mounted = new SessionValue("omp.overlayMounted", () => []);
  var activities = new SessionValue("omp.tabActivity", () => ({}));
  var dismissedGroups = new SessionValue("omp.dismissedGroups", () => []);
  function reportWorking(stored) {
    let working = false;
    for (const tabId in stored) {
      if (stored[tabId]?.activity.state !== "working")
        continue;
      working = true;
      break;
    }
    setBusy("activity", working);
  }
  function isNoReceiver(error) {
    return /Receiving end does not exist/.test(String(error));
  }
  function send(tabId, command) {
    return chrome.tabs.sendMessage(tabId, command, { frameId: 0 });
  }
  async function overlay(tabId, command) {
    try {
      return await send(tabId, command);
    } catch (error) {
      if (!isNoReceiver(error))
        throw error;
    }
    await chrome.scripting.executeScript({ target: { tabId, frameIds: [0] }, files: [OVERLAY_SCRIPT] });
    return await send(tabId, command);
  }
  async function overlayIfPresent(tabId, command, absent) {
    try {
      return await send(tabId, command);
    } catch (error) {
      if (isNoReceiver(error))
        return absent;
      throw error;
    }
  }
  async function relayOverlay(tabId, action) {
    const tabs = await mounted.get();
    switch (action) {
      case "mount":
        await overlay(tabId, { omp: "overlay", op: "mount" });
        if (!tabs.includes(tabId)) {
          tabs.push(tabId);
          mounted.save();
        }
        break;
      case "unmount": {
        const index = tabs.indexOf(tabId);
        if (index >= 0) {
          tabs.splice(index, 1);
          mounted.save();
        }
        await overlayIfPresent(tabId, { omp: "overlay", op: "unmount" }, true);
        break;
      }
      case "hide":
      case "show":
        await overlayIfPresent(tabId, { omp: "overlay", op: action }, true);
        break;
    }
    return {};
  }
  function relayInput(tabId, input) {
    mounted.get().then((tabs) => {
      if (!tabs.includes(tabId))
        return;
      overlay(tabId, { omp: "overlay", op: "input", input }).catch(() => {});
    });
  }
  async function unmountAll() {
    const tabs = await mounted.get();
    const released = tabs.splice(0);
    mounted.save();
    await Promise.allSettled(released.map((tabId) => overlayIfPresent(tabId, { omp: "overlay", op: "unmount" }, true)));
  }
  async function setActivity(tabId, activity) {
    const [stored, dismissed, tab] = await Promise.all([activities.get(), dismissedGroups.get(), chrome.tabs.get(tabId)]);
    const peers = tab.groupId === -1 ? [] : (await chrome.tabs.query({ groupId: tab.groupId })).flatMap((peer) => peer.id === undefined || peer.id === tabId ? [] : [peer.id]);
    const deliveries = [];
    if (activity.state === "idle") {
      delete stored[tabId];
      deliveries.push(overlayIfPresent(tabId, { omp: "overlay", op: "activity", activity, role: "main" }, true));
      for (const peer of peers) {
        if (stored[peer]?.role !== "secondary")
          continue;
        delete stored[peer];
        deliveries.push(overlayIfPresent(peer, { omp: "overlay", op: "activity", activity, role: "secondary" }, true));
      }
      const index = dismissed.indexOf(tab.groupId);
      if (index >= 0) {
        dismissed.splice(index, 1);
        dismissedGroups.save();
      }
    } else {
      stored[tabId] = { activity, role: "main" };
      deliveries.push(overlay(tabId, { omp: "overlay", op: "activity", activity, role: "main" }));
      for (const peer of peers) {
        if (stored[peer]?.role === "main")
          continue;
        stored[peer] = { activity, role: "secondary" };
        if (dismissed.includes(tab.groupId))
          continue;
        deliveries.push(overlay(peer, { omp: "overlay", op: "activity", activity, role: "secondary" }));
      }
    }
    activities.save();
    reportWorking(stored);
    await Promise.allSettled(deliveries);
  }
  async function restoreAfterNavigation(tabId, groupId) {
    const [tabs, stored, dismissed] = await Promise.all([mounted.get(), activities.get(), dismissedGroups.get()]);
    const restores = [];
    if (tabs.includes(tabId))
      restores.push(overlay(tabId, { omp: "overlay", op: "mount" }));
    const shown = stored[tabId];
    if (shown && !(shown.role === "secondary" && dismissed.includes(groupId))) {
      restores.push(overlay(tabId, { omp: "overlay", op: "activity", activity: shown.activity, role: shown.role }));
    }
    await Promise.allSettled(restores);
  }
  async function forgetTab(tabId) {
    const [tabs, stored] = await Promise.all([mounted.get(), activities.get()]);
    const index = tabs.indexOf(tabId);
    if (index >= 0) {
      tabs.splice(index, 1);
      mounted.save();
    }
    if (stored[tabId]) {
      delete stored[tabId];
      activities.save();
      reportWorking(stored);
    }
  }
  function isOverlayEvent(message) {
    return typeof message === "object" && message !== null && "omp" in message && message.omp === "overlay-event" && "op" in message && (message.op === "stop" || message.op === "open" || message.op === "dismiss");
  }
  function installOverlayEvents(deps) {
    activities.get().then(reportWorking);
    chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
      if (!isOverlayEvent(message))
        return;
      const tab = sender.tab;
      if (tab?.id === undefined || sender.id !== chrome.runtime.id)
        return;
      const tabId = tab.id;
      switch (message.op) {
        case "open":
          chrome.sidePanel.open({ windowId: tab.windowId }).catch(() => {});
          deps.emit({ event: "overlay", tabId, action: "open" });
          break;
        case "stop": {
          deps.emit({ event: "overlay", tabId, action: "stop" });
          const toPanel = deps.panelConnected();
          (toPanel ? Promise.resolve(true) : deps.abortTab(tabId)).catch(() => false).then((delivered) => overlayIfPresent(tabId, { omp: "overlay", op: "stopAck", delivered }, true)).catch(() => {});
          break;
        }
        case "dismiss":
          if (tab.groupId !== -1) {
            const groupId = tab.groupId;
            dismissedGroups.get().then((groups) => {
              if (groups.includes(groupId))
                return;
              groups.push(groupId);
              dismissedGroups.save();
            });
          }
          break;
      }
      sendResponse(true);
    });
  }

  function toBase64(bytes) {
    let binary = "";
    for (let start = 0;start < bytes.length; start += 32768) {
      binary += String.fromCharCode(...bytes.subarray(start, start + 32768));
    }
    return btoa(binary);
  }
  async function decode(dataUrl) {
    return await createImageBitmap(await (await fetch(dataUrl)).blob());
  }
  async function captureVisible(windowId) {
    const dataUrl = await chrome.tabs.captureVisibleTab(windowId, { format: "png" });
    const bitmap = await decode(dataUrl);
    try {
      return {
        image: { type: "image", data: dataUrl.slice(dataUrl.indexOf(",") + 1), mimeType: "image/png" },
        width: bitmap.width,
        height: bitmap.height
      };
    } finally {
      bitmap.close();
    }
  }
  async function captureCropped(windowId, rect, viewport) {
    const bitmap = await decode(await chrome.tabs.captureVisibleTab(windowId, { format: "png" }));
    try {
      const scaleX = bitmap.width / viewport.width;
      const scaleY = bitmap.height / viewport.height;
      const left = Math.max(0, Math.round(rect.x * scaleX));
      const top = Math.max(0, Math.round(rect.y * scaleY));
      const width = Math.max(1, Math.min(bitmap.width - left, Math.round(rect.width * scaleX)));
      const height = Math.max(1, Math.min(bitmap.height - top, Math.round(rect.height * scaleY)));
      const canvas = new OffscreenCanvas(width, height);
      const context = canvas.getContext("2d");
      if (!context)
        throw new Error("no 2d canvas in the service worker");
      context.drawImage(bitmap, left, top, width, height, 0, 0, width, height);
      const blob = await canvas.convertToBlob({ type: "image/png" });
      return {
        image: { type: "image", data: toBase64(new Uint8Array(await blob.arrayBuffer())), mimeType: "image/png" },
        width,
        height
      };
    } finally {
      bitmap.close();
    }
  }
  async function withOverlayHidden(tabId, capture) {
    const hidden = await chrome.tabs.sendMessage(tabId, { omp: "overlay", op: "hide" }, { frameId: 0 }).then(() => true).catch(() => false);
    try {
      return await capture();
    } finally {
      if (hidden)
        await chrome.tabs.sendMessage(tabId, { omp: "overlay", op: "show" }, { frameId: 0 }).catch(() => {});
    }
  }
  async function captureTab(tabId, mode) {
    const tab = await chrome.tabs.get(tabId);
    if (!tab.active)
      await chrome.tabs.update(tabId, { active: true });
    const pick = mode === "viewport" ? null : await overlay(tabId, { omp: "overlay", op: "pick", mode });
    if (mode !== "viewport" && !pick)
      return null;
    const png = await withOverlayHidden(tabId, () => pick && pick.kind !== "viewport" ? captureCropped(tab.windowId, pick.rect, pick.viewport) : captureVisible(tab.windowId));
    return {
      ...png,
      tabId,
      url: tab.url ?? tab.pendingUrl ?? "",
      ...pick?.kind === "element" ? { caption: pick.descriptor } : {}
    };
  }

  function installGifExport(ctx) {
    let exporting = Promise.resolve();
    let pending = 0;
    ctx.handle("exportGif", ({ frames }) => {
      if (frames.length === 0)
        return Promise.reject(new Error("There are no screenshots to export"));
      if (pending++ === 0)
        ctx.setBusy("gif", true);
      const result = exporting.then(async () => {
        const contexts = await chrome.runtime.getContexts({
          contextTypes: [chrome.runtime.ContextType.OFFSCREEN_DOCUMENT],
          documentUrls: [chrome.runtime.getURL(OFFSCREEN_PAGE)]
        });
        if (contexts.length === 0) {
          await chrome.offscreen.createDocument({
            url: OFFSCREEN_PAGE,
            reasons: [chrome.offscreen.Reason.BLOBS],
            justification: "Compose and encode a GIF from the run's screenshot frames"
          });
        }
        try {
          const message = { omp: "encode-gif", frames };
          const reply = await chrome.runtime.sendMessage(message);
          if (!reply)
            throw new Error("The GIF encoder did not reply");
          if (!reply.ok)
            throw new Error(reply.error);
          const filename = `omp-run-${new Date().toISOString()}.gif`;
          await chrome.downloads.download({ url: reply.url, filename });
          return { filename };
        } finally {
          await chrome.offscreen.closeDocument();
        }
      }).finally(() => {
        if (--pending === 0)
          ctx.setBusy("gif", false);
      });
      exporting = result.catch(() => {
        return;
      });
      return result;
    });

      target = await chrome.tabs.group({ tabIds: ids, createProperties: { windowId: first.windowId } });
    }

    return { groupId: target };
  }

  class PanelHub {
    #ports = new Map;
    #handlers = new Map;
    #onHello;

      this.#onHello = onHello;
    }
    handle(op, handler) {
      if (this.#handlers.has(op))
        throw new Error(`panel request ${op} already has a handler`);
      this.#handlers.set(op, (request) => isOp(request, op) ? handler(request) : Promise.reject(new Error(`misrouted panel request ${request.op}`)));
    }
    emit(event) {
      for (const port of this.#ports.keys())
        send2(port, { kind: "event", event });
    }
    emitTo(port, event) {
      send2(port, { kind: "event", event });
    }

      return this.#ports.size > 0;
    }
    hasPanel(windowId) {
      for (const panelWindow of this.#ports.values())
        if (panelWindow === windowId)
          return true;
      return false;
    }
    accept(port) {
      if (port.name !== PANEL_PORT)
        return;
      this.#ports.set(port, null);
      port.onMessage.addListener((frame) => {
        if (frame.kind === "hello") {
          this.#ports.set(port, frame.windowId);
          this.#onHello(port, frame.windowId);
          return;
        }
        this.#serve(port, frame.id, frame.request);
      });
      port.onDisconnect.addListener(() => {
        this.#ports.delete(port);
      });
    }
    async#serve(port, id, request) {
      const handler = this.#handlers.get(request.op);
      try {
        if (!handler)
          throw new Error(`no handler for panel request ${request.op}`);
        const result = await handler(request);
        send2(port, { kind: "response", id, ok: true, result });
      } catch (error) {
        send2(port, { kind: "response", id, ok: false, error: error instanceof Error ? error.message : String(error) });
      }
    }
  }
  function send2(port, frame) {
    try {
      port.postMessage(frame);
    } catch {}
  }

  // extension/download-pairing.ts
  var DOWNLOAD_PAIRING_MS = 5000;
  var DOWNLOAD_LOOKUP_MS = 60000;

  class DownloadPairing {
    #now;
    #pairs = new Map;
    #reports = [];
    constructor(now = Date.now) {
      this.#now = now;
    }
    report(guid, url) {
      if (this.#pairs.has(guid) || this.#reports.some((report) => report.guid === guid))
        return;
      this.#dropAgedReports();
      this.#reports.push({ guid, url, at: this.#now() });
    }
    created(id, url) {
      this.#dropAgedReports();
      const index = this.#reports.findIndex((report) => report.url === url);
      if (index < 0)
        return;
      this.#pairs.set(this.#reports[index].guid, { id });
      this.#reports.splice(index, 1);
    }
    #dropAgedReports() {
      const oldest = this.#now() - DOWNLOAD_PAIRING_MS;
      while (this.#reports.length > 0 && this.#reports[0].at < oldest)
        this.#reports.shift();
    }
    finished(id, state) {
      const now = this.#now();
      for (const [guid, pair] of this.#pairs) {
        if (pair.id === id) {
          if (state === "interrupted")
            this.#pairs.delete(guid);
          else
            pair.completedAt = now;
        } else if (pair.completedAt !== undefined && pair.completedAt < now - DOWNLOAD_LOOKUP_MS) {
          this.#pairs.delete(guid);
        }
      }
    }
    idFor(guid) {
      return this.#pairs.get(guid)?.id;
    }
    forget(guid) {
      this.#pairs.delete(guid);
    }
  }

  // extension/foreign-frames.ts
  var RETRY_DELAYS_MS = [50, 100, 200, 400];
  var sweeps = new Map;
  async function pastForeignFrames(tabId, request) {
    for (let attempt = 0;; attempt++) {
      try {
        return await request();
      } catch (error) {
        if (attempt === RETRY_DELAYS_MS.length || !FOREIGN_FRAME_REFUSAL.test(String(error)) || await sweep(tabId) === 0)
          throw error;
      }
      const { promise, resolve } = Promise.withResolvers();
      setTimeout(resolve, RETRY_DELAYS_MS[attempt]);
      await promise;
    }
  }
  function sweep(tabId) {
    let pending = sweeps.get(tabId);
    if (!pending) {
      pending = neutralizeForeignFrames(tabId).finally(() => sweeps.delete(tabId));
      sweeps.set(tabId, pending);
    }
    return pending;
  }
  var reportHidden = () => {};
  function onForeignFramesHidden(report) {
    reportHidden = report;
  }
  var MAX_PATH_LENGTH = 100;
  var MAX_TITLE_LENGTH = 60;
  var PROBE_TIMEOUT_MS = 500;
  var PROBE_MAX_BYTES = 64 * 1024;
  async function reportEmptied(tabId, srcsByFrame) {
    const probes = await Promise.allSettled(Array.from(srcsByFrame, ([frameId, srcs]) => chrome.scripting.executeScript({
      target: { tabId, frameIds: [frameId] },
      world: "ISOLATED",
      injectImmediately: true,
      args: [srcs, PROBE_TIMEOUT_MS, PROBE_MAX_BYTES],
      func: async (urls, timeoutMs, maxBytes) => await Promise.all(urls.map(async (url) => {
        try {
          const response = await fetch(url, { signal: AbortSignal.timeout(timeoutMs) });
          if (!response.ok)
            return null;
          const reader = response.body?.getReader();
          const decoder = new TextDecoder;
          let head = "";
          for (let read = 0;reader && read < maxBytes && !/<\/title>/i.test(head); ) {
            const chunk = await reader.read();
            if (chunk.done)
              break;
            read += chunk.value.byteLength;
            head += decoder.decode(chunk.value, { stream: true });
          }
          reader?.cancel().catch(() => {});
          const title = /<title[^>]*>([^<]*)<\/title>/i.exec(head.slice(0, maxBytes))?.[1]?.trim();
          return { url: response.url, title };
        } catch {
          return null;
        }
      }))
    })));
    const frames = new Map;
    for (const probe of probes) {
      if (probe.status === "rejected")
        continue;
      for (const loaded of probe.value.flatMap(({ result }) => result ?? [])) {
        if (!loaded?.url.startsWith("chrome-extension://"))
          continue;
        const { host: extensionId, pathname } = new URL(loaded.url);
        if (!/^[a-p]{32}$/.test(extensionId) || extensionId === chrome.runtime.id)
          continue;
        const path = pathname.slice(1, 1 + MAX_PATH_LENGTH);
        const title = loaded.title?.slice(0, MAX_TITLE_LENGTH);
        frames.set(`${extensionId}/${path}`, { extensionId, path, ...title ? { title } : {} });
      }
    }
    if (frames.size > 0)
      reportHidden(tabId, [...frames.values()]);
  }
  async function neutralizeForeignFrames(tabId) {
    let results;
    try {
      results = await chrome.scripting.executeScript({
        target: { tabId, allFrames: true },
        injectImmediately: true,
        args: [chrome.runtime.id],
        func: (self) => {
          const swept = { found: 0, emptied: [] };
          const documentKey = "document";
          const page = Reflect.get(globalThis, documentKey);
          const dom = chrome.dom;
          const visit = (root) => {
            const walker = page.createTreeWalker(root, 1);
            for (let element = walker.nextNode();element; element = walker.nextNode()) {
              const tag = element.tagName.toUpperCase();
              if (tag === "IFRAME" || tag === "FRAME") {
                const src = element.getAttribute("src") ?? "";
                if (src.startsWith("chrome-extension://") && !src.startsWith(`chrome-extension://${self}/`)) {
                  if (tag === "FRAME") {
                    element.setAttribute("src", "about:blank");
                    swept.emptied.push(src);
                  } else if (element.getAttribute("srcdoc") === null) {
                    element.setAttribute("srcdoc", "");
                    swept.emptied.push(src);
                  }
                  element.style.setProperty("display", "none", "important");
                  swept.found++;
                }
              }
              let shadow = null;
              try {
                shadow = dom.openOrClosedShadowRoot(element);
              } catch {}
              if (shadow)
                visit(shadow);
            }
          };
          visit(page);
          return swept;
        }
      });
    } catch {
      return 0;
    }
    let found = 0;
    const srcsByFrame = new Map;
    for (const { frameId, result } of results) {
      found += result?.found ?? 0;
      if (result && result.emptied.length > 0)
        srcsByFrame.set(frameId, result.emptied);
    }
    if (srcsByFrame.size > 0)
      reportEmptied(tabId, srcsByFrame).catch(() => {});
    return found;
  }

  // extension/worker/relay-rpc.ts
  async function runRelayRpc(request, origin) {
    switch (request.op) {
      case "attach": {
        const attach = () => chrome.debugger.attach({ tabId: request.tabId }, "1.3");
        await (request.clearForeignFrames ? pastForeignFrames(request.tabId, attach) : attach());
        return {};
      }
      case "detach":
        relayInitiatedDetachTabs.add(request.tabId);
        try {
          await chrome.debugger.detach({ tabId: request.tabId });
          return {};
        } catch (error) {
          relayInitiatedDetachTabs.delete(request.tabId);
          throw error;
        }
      case "send": {
        const send = () => chrome.debugger.sendCommand(request.sessionId ? { tabId: request.tabId, sessionId: request.sessionId } : { tabId: request.tabId }, request.method, request.params);
        return await (request.clearForeignFrames ? pastForeignFrames(request.tabId, send) : send());
      }
      case "createTab": {
        const tab = await chrome.tabs.create({ url: request.url, active: request.active ?? true });
        const snap = snapshot(tab);
        if (!snap)
          throw new Error("created tab has no id");
        if (!origin.open)
          removeTab(snap.tabId).catch(() => {});
        return { tab: snap };
      }
      case "removeTab":
        await removeTab(request.tabId);
        return {};
      case "activateTab":
        await focusTab(request.tabId);
        return {};
      case "group":
        return await enqueueGroupOp(() => groupRelayTabs(request.tabIds, request.title, request.color));
      case "ungroup": {
        const { tabIds } = request;
        if (isNonEmpty(tabIds))
          await enqueueGroupOp(() => chrome.tabs.ungroup(tabIds).catch(() => {}));
        return {};
      }
      case "findDownload":
        return await findDownload(request.guid);
      case "overlay":
        return await relayOverlay(request.tabId, request.action);
    }
  }

  // extension/worker/rpc-sessions.ts
  var READY_TIMEOUT_MS = 30000;
  var RPC_ABORT_LINE = JSON.stringify({ type: "abort" });

  class RpcSessions {
    #list = new SessionValue("omp.rpcSessions", () => []);
    #live = new Map;
    #lineListeners = new Set;
    #changed;
    #panelLine;
    #reconciled = null;
    constructor(changed, panelLine) {
      this.#changed = changed;
      this.#panelLine = panelLine;
    }
    list() {
      this.#reconciled ??= this.#list.get().then((sessions) => {
        let stale = false;
        for (const session of sessions) {
          if (session.exited || this.#live.has(session.sessionKey))
            continue;
          session.exited = { code: null, stderr: "the extension's service worker restarted" };
          stale = true;
        }
        if (stale)
          this.#list.save();
        return sessions;
      });
      return this.#reconciled;
    }
    onLine(listener) {
      this.#lineListeners.add(listener);
      return () => this.#lineListeners.delete(listener);
    }
    async forTab(tabId) {
      const sessions = await this.list();
      return sessions.find((session) => !session.exited && this.#live.has(session.sessionKey) && session.tabIds.includes(tabId));
    }
    async open(options) {
      const sessions = await this.list();
      const info = {
        sessionKey: crypto.randomUUID(),
        cwd: options.cwd,
        title: options.title,
        startedAt: Date.now(),
        tabIds: [...options.tabIds]
      };
      const port = chrome.runtime.connectNative(NATIVE_HOST);
      const live = { port, chunks: new Map };
      this.#live.set(info.sessionKey, live);
      const ready = Promise.withResolvers();
      let started = false;
      let relay;
      const timer = setTimeout(() => {
        ready.reject(new Error(`omp native host did not start omp within ${READY_TIMEOUT_MS / 1000}s`));
        port.disconnect();
        this.#live.delete(info.sessionKey);
      }, READY_TIMEOUT_MS);
      const exit = (code, stderr) => {
        if (!this.#live.has(info.sessionKey))
          return;
        this.#live.delete(info.sessionKey);
        clearTimeout(timer);
        if (!started) {
          ready.reject(new Error(stderr.trim() || `omp native host exited (${code ?? "disconnected"})`));
          return;
        }
        info.exited = { code, stderr };
        this.#list.save();
        this.#changed(sessions);
        this.#ended(info.sessionKey, code, stderr);
      };
      port.onMessage.addListener((message) => {
        switch (message.t) {
          case "linked":
            relay = connectChatRelay(port);
            return;
          case "relay":
            relay?.receive(message.line);
            return;
          case "ready":
            started = true;
            clearTimeout(timer);
            ready.resolve();
            return;
          case "rpc":
            this.#deliver(info.sessionKey, message.line);
            return;
          case "chunk": {
            const parts = live.chunks.get(message.id) ?? [];
            parts[message.index] = message.data;
            if (!message.last) {
              live.chunks.set(message.id, parts);
              return;
            }
            live.chunks.delete(message.id);
            const line = parts.join("");
            if (message.relay)
              relay?.receive(line);
            else
              this.#deliver(info.sessionKey, line);
            return;
          }
          case "exit":
            exit(message.code, message.stderr);
            port.disconnect();
            return;
        }
      });
      port.onDisconnect.addListener(() => {
        relay?.close();
        exit(null, chrome.runtime.lastError?.message ?? "omp native host disconnected");
      });
      const open = { t: "open", cwd: options.cwd, extensionId: chrome.runtime.id };
      port.postMessage(open);
      await ready.promise;
      sessions.push(info);
      this.#list.save();
      this.#changed(sessions);
      return info;
    }
    send(sessionKey, line) {
      const live = this.#live.get(sessionKey);
      if (!live)
        throw new Error(`rpc session ${sessionKey} is not running`);
      const frame = { t: "rpc", line };
      live.port.postMessage(frame);
    }
    async close(sessionKey) {
      const sessions = await this.list();
      const live = this.#live.get(sessionKey);
      if (live) {
        this.#live.delete(sessionKey);
        live.port.disconnect();
        this.#ended(sessionKey, null, "closed");
      }
      const index = sessions.findIndex((session) => session.sessionKey === sessionKey);
      if (index < 0)
        return;
      sessions.splice(index, 1);
      this.#list.save();
      this.#changed(sessions);
    }
    #deliver(sessionKey, line) {
      this.#panelLine(sessionKey, line);
      this.#notify(sessionKey, line);
    }
    #ended(sessionKey, code, stderr) {
      this.#notify(sessionKey, JSON.stringify({ type: "rpc_exit", code, stderr }));
    }
    #notify(sessionKey, line) {
      for (const listener of this.#lineListeners) {
        try {
          listener(sessionKey, line);
        } catch {}
      }
    }
  }

  function isRecord(value) {
    return !!value && typeof value === "object" && !Array.isArray(value);
  }

  // extension/panel/tab-context.ts
  var TAB_AUTHORIZATION = "This is my tab; navigate, click and type in it as needed.";
  function escapeAttr(value) {
    return value.replaceAll("&", "&amp;").replaceAll('"', "&quot;").replaceAll("<", "&lt;").replaceAll(">", "&gt;").replace(/\s+/g, " ");
  }
  async function tabViewport(tab) {
    if (tab.viewport !== null)
      return tab.viewport;
    try {
      const [info, zoom] = await Promise.all([chrome.tabs.get(tab.tabId), chrome.tabs.getZoom(tab.tabId)]);
      if (info.width === undefined || info.height === undefined)
        return null;
      const factor = zoom > 0 ? zoom : 1;
      return { width: Math.round(info.width / factor), height: Math.round(info.height / factor) };
    } catch {
      return null;
    }
  }
  function tabContextBlock(tab, viewport) {
    const attrs = [
      `target="${escapeAttr(tab.targetId ?? "")}"`,
      `url="${escapeAttr(tab.url)}"`,
      `title="${escapeAttr(tab.title)}"`
    ];
    if (viewport !== null)
      attrs.push(`viewport="${viewport.width}x${viewport.height}"`);
    return `<chrome-tab ${attrs.join(" ")}/>
${TAB_AUTHORIZATION}`;
  }
  function composePrompt(blocks, text) {
    return blocks.length === 0 ? text : `${blocks.join(`
`)}

${text}`;
  }

  // extension/worker/schedules.ts
  var ALARM_PREFIX = "omp-shortcut:";
  var NOTICE_PREFIX = "omp-shortcut-result:";
  function wallTime(year, month, day, hour, minute) {
    if (year < 0 || year >= 100)
      return +new Date(year, month, day, hour, minute);
    const date = new Date(year + 400, month, day, hour, minute);
    date.setFullYear(date.getFullYear() - 400);
    return +date;
  }
  function monthLength(year, month) {
    const end = new Date(0);
    end.setUTCFullYear(year, month + 1, 0);
    return end.getUTCDate();
  }
  function nextRunAt(schedule, after) {
    if (!/^([01]\d|2[0-3]):[0-5]\d$/.test(schedule.time) || !Number.isFinite(after))
      return;
    const hour = Number(schedule.time.slice(0, 2));
    const minute = Number(schedule.time.slice(3));
    const now = new Date(after);
    const year = now.getFullYear();
    const month = now.getMonth();
    const day = now.getDate();
    const at = (y, m, d) => wallTime(y, m, d, hour, minute);
    const clamped = (y, m, d) => at(y, m, Math.min(d, monthLength(y, m)));
    switch (schedule.kind) {
      case "once": {
        if (!/^\d{4,}-\d{2}-\d{2}$/.test(schedule.date))
          return;
        const y = Number(schedule.date.slice(0, -6));
        const m = Number(schedule.date.slice(-5, -3)) - 1;
        const d = Number(schedule.date.slice(-2));
        if (y < 1 || m < 0 || m > 11 || d < 1 || d > monthLength(y, m))
          return;
        const runAt = at(y, m, d);
        return runAt > after ? runAt : undefined;
      }
      case "daily": {
        const today = at(year, month, day);
        return today > after ? today : at(year, month, day + 1);
      }
      case "weekly": {
        const daysAhead = (schedule.weekday - now.getDay() + 7) % 7;
        const target = at(year, month, day + daysAhead);
        return target > after ? target : at(year, month, day + daysAhead + 7);
      }
      case "monthly": {
        const target = clamped(year, month, schedule.day);
        return target > after ? target : clamped(year, month + 1, schedule.day);
      }
      case "yearly": {
        const targetMonth = schedule.month - 1;
        const target = clamped(year, targetMonth, schedule.day);
        return target > after ? target : clamped(year + 1, targetMonth, schedule.day);
      }
    }
  }
  function planSchedule(schedule, now, storedAt) {
    return { due: storedAt !== undefined && storedAt <= now, nextAt: nextRunAt(schedule, now) };
  }
  function recurrence(schedule) {
    if (!schedule)
      return;
    const time = `${schedule.kind}:${schedule.time}`;
    switch (schedule.kind) {
      case "once":
        return `${time}:${schedule.date}`;
      case "daily":
        return time;
      case "weekly":
        return `${time}:${schedule.weekday}`;
      case "monthly":
        return `${time}:${schedule.day}`;
      case "yearly":
        return `${time}:${schedule.month}:${schedule.day}`;
    }
  }
  async function readShortcuts() {
    const { [SHORTCUTS_KEY]: shortcuts = [] } = await chrome.storage.local.get(SHORTCUTS_KEY);
    return shortcuts;
  }
  function errorMessage(error) {
    return error instanceof Error ? error.message : String(error);
  }
  function command(ctx, key, cmd) {
    const result = Promise.withResolvers();
    const unsubscribe = ctx.onRpcLine((sessionKey, line) => {
      if (sessionKey !== key || !line.includes(cmd.id) && !line.includes('"rpc_exit"'))
        return;
      let frame;
      try {
        frame = JSON.parse(line);
      } catch {
        return;
      }
      if (!isRecord(frame))
        return;
      if (frame.type === "rpc_exit") {
        const stderr = typeof frame.stderr === "string" ? frame.stderr.trim() : "";
        result.reject(new Error(stderr || "omp exited before the shortcut finished"));
        return;
      }
      if (frame.id !== cmd.id)
        return;
      if (frame.type === "response" && frame.command === cmd.type) {
        if (frame.success === false) {
          result.reject(new Error(typeof frame.error === "string" ? frame.error : `${cmd.type} failed`));
        } else if (cmd.type === "set_model" || isRecord(frame.data) && frame.data.agentInvoked === false) {
          result.resolve();
        }
      } else if (frame.type === "prompt_result" && cmd.type === "prompt") {
        if (frame.status === "completed")
          result.resolve();
        else if (frame.status === "aborted")
          result.reject(new Error("Shortcut was stopped"));
        else {
          const detail = isRecord(frame.error) && typeof frame.error.message === "string" ? frame.error.message : "Prompt failed";
          result.reject(new Error(detail));
        }
      }
    });
    try {
      ctx.sendRpc(key, JSON.stringify(cmd));
    } catch (error) {
      unsubscribe();
      throw error;
    }
    return result.promise.finally(unsubscribe);
  }
  function installSchedules(ctx) {
    let storageTail = Promise.resolve();
    function serialize(job) {
      const result = storageTail.then(job);
      storageTail = result.then(() => {
        return;
      }, () => {
        return;
      });
      return result;
    }
    async function complete(shortcut, startedAt, tab, failure) {
      try {
        if (tab?.id !== undefined)
          await ctx.setActivity(tab.id, { state: "idle" });
      } finally {
        await Promise.all([
          serialize(async () => {
            const shortcuts = await readShortcuts();
            const current = shortcuts.find((item) => item.id === shortcut.id);
            if (!current || (current.lastRunAt ?? 0) > startedAt)
              return;
            current.lastRunAt = startedAt;
            if (failure === undefined)
              delete current.lastRunError;
            else
              current.lastRunError = failure;
            await chrome.storage.local.set({ [SHORTCUTS_KEY]: shortcuts });
          }),
          chrome.notifications.create(`${NOTICE_PREFIX}${tab?.id ?? "none"}:${crypto.randomUUID()}`, {
            type: "basic",
            iconUrl: "icons/omp-128.png",
            title: failure === undefined ? `omp · /${shortcut.name} finished` : `omp · /${shortcut.name} failed`,
            message: failure === undefined ? "Click to open the session." : failure.slice(0, 250)
          })
        ]);
      }
    }
    function report(error) {
      console.error("omp shortcut:", error);
    }
    async function run(shortcut) {
      const startedAt = Date.now();
      let tab;
      try {
        const { [SETTINGS_KEY]: settings } = await chrome.storage.local.get(SETTINGS_KEY);
        const startUrl = shortcut.startUrl || "about:blank";
        tab = await chrome.tabs.create({ url: startUrl, active: false });
        const tabId = tab.id;
        if (tabId === undefined)
          throw new Error("Chrome did not assign the shortcut tab an id");
        await enqueueGroupOp(() => groupPanelTabs([tabId], `omp · ${shortcut.name}`));
        const context = await ctx.tabContext(tabId);
        if (!context.targetId)
          throw new Error("Chrome has not initialized the omp tab target");
        if (!context.url || context.url === "about:blank")
          context.url = tab.pendingUrl || startUrl;
        const block = tabContextBlock(context, await tabViewport(context));
        const message = composePrompt([block], shortcut.prompt);
        const session = await ctx.openRpcSession({
          cwd: shortcut.cwd ?? settings?.defaultCwd ?? "~",
          title: shortcut.name,
          tabIds: [tabId]
        });
        if (shortcut.model) {
          const slash = shortcut.model.indexOf("/");
          if (slash < 1 || slash === shortcut.model.length - 1)
            throw new Error("Shortcut model must be provider/id");
          await command(ctx, session.sessionKey, {
            type: "set_model",
            id: crypto.randomUUID(),
            provider: shortcut.model.slice(0, slash),
            modelId: shortcut.model.slice(slash + 1)
          });
        }
        await ctx.setActivity(tabId, { state: "working", session: session.sessionKey, canStop: true });
        command(ctx, session.sessionKey, { type: "prompt", id: crypto.randomUUID(), message }).then(() => complete(shortcut, startedAt, tab), (error) => complete(shortcut, startedAt, tab, errorMessage(error))).catch(report);
        ctx.emit({ event: "scheduledRun", shortcutId: shortcut.id, session });
        return session;
      } catch (error) {
        await complete(shortcut, startedAt, tab, errorMessage(error)).catch(report);
        throw error;
      }
    }
    function reconcile(edited = new Set) {
      serialize(async () => {
        const shortcuts = await readShortcuts();
        const now = Date.now();
        const due = [];
        const desired = new Map;
        let changed = false;
        for (const shortcut of shortcuts) {
          const last = edited.has(shortcut.id) ? undefined : shortcut.nextRunAt;
          const plan = shortcut.schedule ? planSchedule(shortcut.schedule, now, last) : undefined;
          if (plan?.due)
            due.push(shortcut);
          const next = plan?.nextAt;
          if (next !== undefined)
            desired.set(`${ALARM_PREFIX}${shortcut.id}`, next);
          if (shortcut.nextRunAt === next)
            continue;
          changed = true;
          if (next === undefined)
            delete shortcut.nextRunAt;
          else
            shortcut.nextRunAt = next;
        }
        if (changed)
          await chrome.storage.local.set({ [SHORTCUTS_KEY]: shortcuts });
        for (const shortcut of due)
          run(shortcut).catch(report);
        const alarms = await chrome.alarms.getAll();
        for (const alarm of alarms) {
          if (!alarm.name.startsWith(ALARM_PREFIX))
            continue;
          const when = desired.get(alarm.name);
          if (when === undefined)
            await chrome.alarms.clear(alarm.name);
          else if (when === alarm.scheduledTime)
            desired.delete(alarm.name);
        }
        for (const [name, when] of desired)
          await chrome.alarms.create(name, { when });
      }).catch(report);
    }
    ctx.handle("runShortcut", async ({ shortcutId }) => {
      const shortcut = (await readShortcuts()).find((item) => item.id === shortcutId);
      if (!shortcut)
        throw new Error("Shortcut not found");
      return run(shortcut);
    });
    chrome.alarms.onAlarm.addListener((alarm) => {
      if (alarm.name.startsWith(ALARM_PREFIX))
        reconcile();
    });
    chrome.runtime.onStartup.addListener(() => reconcile());
    chrome.storage.local.onChanged.addListener((changes) => {
      const change = changes[SHORTCUTS_KEY];
      if (!change)
        return;
      const before = Array.isArray(change.oldValue) ? change.oldValue : [];
      const after = Array.isArray(change.newValue) ? change.newValue : [];
      const previous = new Map(before.map((shortcut) => [shortcut.id, recurrence(shortcut.schedule)]));
      const edited = new Set(after.filter((shortcut) => recurrence(shortcut.schedule) !== previous.get(shortcut.id)).map((shortcut) => shortcut.id));
      if (edited.size || before.length !== after.length)
        reconcile(edited);
    });
    chrome.notifications.onClicked.addListener((id) => {
      if (!id.startsWith(NOTICE_PREFIX))
        return;
      const tabId = Number(id.slice(NOTICE_PREFIX.length).split(":", 1)[0]);
      if (Number.isInteger(tabId)) {
        const selected = chrome.tabs.update(tabId, { active: true });
        chrome.sidePanel.open({ tabId }).catch(report);
        selected.then((tab) => tab ? chrome.windows.update(tab.windowId, { focused: true }) : undefined).catch(report);
      }
      chrome.notifications.clear(id);
    });
    reconcile();
  }

  var TOGGLE_COMMAND = "toggle-side-panel";
  function toggle(windowId, isOpen) {
    if (isOpen(windowId) && typeof chrome.sidePanel.close === "function") {
      chrome.sidePanel.close({ windowId }).catch(() => {});
      return;
    }
    chrome.sidePanel.open({ windowId }).catch(() => {});
  }
  function installSidePanel(isOpen) {
    chrome.sidePanel.setPanelBehavior({ openPanelOnActionClick: true }).catch(() => {});
    chrome.commands.onCommand.addListener((command, tab) => {
      if (command !== TOGGLE_COMMAND)
        return;
      if (tab) {
        toggle(tab.windowId, isOpen);
        return;
      }
      chrome.windows.getLastFocused().then((window) => {
        if (window.id !== undefined)
          toggle(window.id, isOpen);
      });
    });
  }

  // extension/worker/teach.ts
  var CAPTURE_INTERVAL_MS = 510;
  async function startRecorder(tabId) {
    const [main] = await chrome.scripting.executeScript({
      target: { tabId },
      files: ["recorder.js"],
      injectImmediately: true
    });
    if (!main)
      throw new Error("Could not start Teach in this tab");
    const command = { omp: "teach", op: "start" };
    const started = await chrome.tabs.sendMessage(tabId, command, { documentId: main.documentId });
    if (started !== true)
      throw new Error("Teach recorder did not start");
  }
  function installTeach(ctx) {
    let recording;
    function append(state, step) {
      state.steps.push(step);
      state.captures = state.captures.then(async () => {
        try {
          if (step.kind === "navigate") {
            const current = await ctx.tabContext(state.tabId);
            if (current.url === step.url) {
              step.title = current.title;
              if (current.viewport)
                step.viewport = current.viewport;
            }
          }
          const wait = state.lastCaptureAt + CAPTURE_INTERVAL_MS - Date.now();
          if (wait > 0) {
            const timer = Promise.withResolvers();
            setTimeout(timer.resolve, wait);
            await timer.promise;
          }
          const tab = await chrome.tabs.get(state.tabId);
          if (tab.active) {
            state.lastCaptureAt = Date.now();
            const screenshot = await ctx.captureVisible(tab.windowId);
            if ((await chrome.tabs.get(state.tabId)).active)
              step.screenshot = screenshot.image;
          }
        } catch {}
        ctx.emit({ event: "teachStep", step });
      });
    }
    ctx.handle("teachStart", async ({ tabId }) => {
      if (recording)
        throw new Error("A Teach recording is already running");
      const state = {
        tabId,
        url: "",
        viewport: { width: 0, height: 0 },
        steps: [],
        lastCaptureAt: 0,
        installation: Promise.resolve(),
        captures: Promise.resolve()
      };
      recording = state;
      ctx.setBusy("teach", true);
      state.installation = (async () => {
        const tab = await chrome.tabs.get(tabId);
        state.url ||= tab.url ?? "";
        state.viewport = { width: tab.width ?? 0, height: tab.height ?? 0 };
        await startRecorder(tabId);
      })();
      try {
        await state.installation;
      } catch (error) {
        if (recording === state) {
          recording = undefined;
          ctx.setBusy("teach", false);
        }
        throw error;
      }
    });
    ctx.handle("teachStop", async () => {
      const state = recording;
      if (!state)
        return [];
      state.stop ??= (async () => {
        await state.installation.catch(() => {
          return;
        });
        const command = { omp: "teach", op: "stop" };
        await chrome.tabs.sendMessage(state.tabId, command, { frameId: 0 }).catch(() => {
          return;
        });
        try {
          await state.captures;
          return state.steps;
        } finally {
          if (recording === state) {
            recording = undefined;
            ctx.setBusy("teach", false);
          }
        }
      })();
      return state.stop;
    });
    chrome.runtime.onMessage.addListener((message, sender) => {
      const state = recording;
      if (message?.omp !== "teach-step" || !state || sender.tab?.id !== state.tabId || sender.frameId !== 0)
        return;
      state.viewport = message.step.viewport;
      append(state, message.step);
    });
    chrome.tabs.onRemoved.addListener((tabId) => {
      if (recording?.tabId === tabId)
        ctx.setBusy("teach", false);
    });
    chrome.tabs.onUpdated.addListener((tabId, change, tab) => {
      const state = recording;
      if (!state || state.stop || tabId !== state.tabId)
        return;
      if (change.url !== undefined && change.url !== state.url) {
        state.url = change.url;
        append(state, {
          kind: "navigate",
          url: change.url,
          title: tab.title ?? "",
          at: Date.now(),
          viewport: { width: tab.width ?? state.viewport.width, height: tab.height ?? state.viewport.height }
        });
      }
      if (change.url !== undefined || change.status === "complete") {
        state.installation = state.installation.catch(() => {
          return;
        }).then(() => {
          if (!state.stop && recording === state)
            return startRecorder(tabId);
        }).catch(() => {
          return;
        });
      }
    });
  }
